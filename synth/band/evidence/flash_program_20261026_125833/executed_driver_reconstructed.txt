"""Program only the selected, hash-checked board release. Default: offline preflight.

--execute replaces the existing boot image; no backup is implied. A successful
vendor verify and JTAG reload do not constitute physical cold-start acceptance.
"""
import argparse
import datetime
import hashlib
import json
from pathlib import Path
import re
import subprocess
from verify_fs import verify_fs

BAND = Path(__file__).resolve().parents[1]

def preflight():
    selected = json.loads((BAND / 'CURRENT_RELEASE.json').read_text(encoding='utf-8'))
    name = selected['release']
    if Path(name).name != name or '/' in name or '\\' in name:
        raise ValueError('Release must be a local directory name')
    release = BAND / 'release' / name
    manifest = json.loads((release / 'manifest.json').read_text(encoding='utf-8'))
    bitstream = release / 'starcluster_band.fs'
    verify_fs(bitstream)
    digest = hashlib.sha256(bitstream.read_bytes()).hexdigest()
    if digest != selected['bitstream_sha256'] or digest != manifest['bitstream_sha256']:
        raise ValueError('Selected bitstream hash mismatch')
    if manifest.get('build_id') != selected.get('build_id') or not manifest.get('build_id'):
        raise ValueError('Build identity mismatch')
    if manifest.get('top_module') != 'board_pt8211_dvi_top':
        raise ValueError('This loader requires the reviewed fixed-pin PT8211 board shell')
    if manifest.get('setup_violations') != 0 or manifest.get('hold_violations') != 0:
        raise ValueError('Missing clean timing')
    sources = manifest.get('snapshot_sha256', {})
    if not sources or 'board.cst' not in sources or 'board.sdc' not in sources:
        raise ValueError('Missing frozen constraints')
    for source, expected in sources.items():
        path = (release / 'source_snapshot' / source).resolve()
        if not path.is_relative_to((release / 'source_snapshot').resolve()):
            raise ValueError('Invalid snapshot path')
        if hashlib.sha256(path.read_bytes()).hexdigest() != expected:
            raise ValueError('Snapshot changed: ' + source)
    return bitstream, dict(release=name, build_id=manifest['build_id'],
                          bitstream_sha256=digest, snapshot_files=len(sources))

def main():
    ap = argparse.ArgumentParser(description=__doc__)
    ap.add_argument('--execute', action='store_true')
    ap.add_argument('--location', type=int, default=0)
    ap.add_argument('--channel', type=int, default=0)
    ap.add_argument('--programmer', default='D:/Gowin/Gowin_V1.9.12.03_x64/Programmer/bin/programmer_cli.exe')
    args = ap.parse_args()
    bitstream, record = preflight()
    record.update(status='preflight_passed', flash_written=False,
                  previous_flash_backup_available=False, cold_start_verified=False)
    if not args.execute:
        print(json.dumps(record, indent=2))
        return
    logdir = BAND / 'evidence' / ('flash_program_' + datetime.datetime.now().strftime('%Y%m%d_%H%M%S'))
    logdir.mkdir()
    record['started'] = datetime.datetime.now().isoformat()
    def save():
        (logdir / 'result.json').write_text(json.dumps(record, indent=2), encoding='utf-8')
    def run(name, arguments, timeout, success):
        command = [args.programmer, *arguments]
        (logdir / (name + '.command.json')).write_text(json.dumps(command), encoding='utf-8')
        with (logdir / (name + '.out.txt')).open('wb') as out, (logdir / (name + '.err.txt')).open('wb') as err:
            result = subprocess.run(command, stdout=out, stderr=err, timeout=timeout,
                                    creationflags=getattr(subprocess, 'CREATE_NO_WINDOW', 0))
        text = (logdir / (name + '.out.txt')).read_text(errors='replace') + (logdir / (name + '.err.txt')).read_text(errors='replace')
        if result.returncode or re.search(r'\b(error|failed|failure)\b', text, re.I) or not re.search(success, text, re.I):
            raise RuntimeError('Programmer operation failed: ' + name + '; inspect ' + str(logdir))
    common = ['--device', 'GW5AST-138B', '--cable-index', '4', '--location', str(args.location), '--channel', str(args.channel), '--frequency', '0.5MHz']
    save()
    try:
        run('probe', common + ['--run', '0'], 20, r'success|IDCODE|ID code')
        record['status'] = 'programming_flash'; record['flash_may_be_modified'] = True; save()
        print('Programming and verifying selected boot image. Logs: ' + str(logdir), flush=True)
        run('program_verify', common + ['--run', '54', '--fsFile', str(bitstream)], 600, r'Program and Verify Flash successfully')
        record.update(status='flash_verified', flash_written=True); save()
        run('reload', common + ['--run', '1'], 30, r'success|Finished\.')
        record['status'] = 'flash_verified_reload_returned_pending_uart'
    except Exception as error:
        record.update(status='failed', error=str(error))
        raise
    finally:
        record['finished'] = datetime.datetime.now().isoformat(); save()
    print(json.dumps(record, indent=2))

if __name__ == '__main__':
    main()
